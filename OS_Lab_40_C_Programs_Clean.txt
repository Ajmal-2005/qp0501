4. Shortest Job First Scheduling
#include <stdio.h>
int main(){
    int n,i,j,b[20],p[20],t;
    scanf("%d",&n); for(i=0;i<n;i++){p[i]=i+1;scanf("%d",&b[i]);}
    for(i=0;i<n-1;i++)for(j=i+1;j<n;j++)if(b[i]>b[j]){

}

6. Preemptive Priority Scheduling
#include <stdio.h>
int main(){
    int n,i,t=0,done=0,bt[20],rem[20],pr[20],wt[20]={0},tat[20],best;
    scanf("%d",&n);for(i=0;i<n;i++){scanf("%d%d",&bt[i],&pr[i]);rem[i]=bt[i];}
    while(done<n){best=-1;for(i=0;i<n;i++)if(rem[i]&& (best<0||pr[i]<pr[best]))best=i;
        if(best<0)break;rem[best]--;t++;if(!rem[best]){tat[best]=t;done++;}
    }
    for(i=0;i<n;i++)printf("P%d WT=%d TAT=%d\n",i+1,tat[i]-bt[i],tat[i]);
    return 0;
}

17. Banker's Algorithm
#include <stdio.h>
int main(){int n,m,a[10][10],max[10][10],need[10][10],avail[10],finish[10]={0},safe[10],i,j,k,c=0,ok;
scanf("%d%d",&n,&m);for(i=0;i<n;i++)for(j=0;j<m;j++)scanf("%d",&a[i][j]);
for(i=0;i<n;i++)for(j=0;j<m;j++)scanf("%d",&max[i][j]);for(j=0;j<m;j++)scanf("%d",&avail[j]);
for(i=0;i<n;i++)for(j=0;j<m;j++)need[i][j]=max[i][j]-a[i][j];
while(c<n){ok=0;for(i=0;i<n;i++)if(!finish[i]){for(j=0;j<m;j++)if(need[i][j]>avail[j])break;
if(j==m){for(k=0;k<m;k++)avail[k]+=a[i][k];safe[c++]=i;finish[i]=1;ok=1;}}if(!ok)break;}
if(c==n){printf("Safe sequence: ");for(i=0;i<n;i++)printf("P%d ",safe[i]);}else printf("Unsafe");return 0;}

18. Producer Consumer Using Semaphores
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
int buffer[5],in=0,out=0;sem_t empty,full,mutex;
void *producer(void *x){for(int i=1;i<=5;i++){sem_wait(&empty);sem_wait(&mutex);buffer[in++%5]=i;printf("Produced %d\n",i);sem_post(&mutex);sem_post(&full);}return 0;}
void *consumer(void *x){for(int i=1;i<=5;i++){sem_wait(&full);sem_wait(&mutex);printf("Consumed %d\n",buffer[out++%5]);sem_post(&mutex);sem_post(&empty);}return 0;}
int main(){pthread_t p,c;sem_init(&empty,0,5);sem_init(&full,0,0);sem_init(&mutex,0,1);pthread_create(&p,0,producer,0);pthread_create(&c,0,consumer,0);pthread_join(p,0);pthread_join(c,0);return 0;}
